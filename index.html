Change this to a strong password YOUR_IP=$(curl -s ifconfig.me) # Auto-detect your public IP BACKUP_DIR="/root/lockdown_backup" LOG_FILE="/root/lockdown.log" DATE=$(date '+%Y-%m-%d_%H-%M-%S') # --- COLORS --- RED='\033[0;31m' GREEN='\033[0;32m' YELLOW='\033[1;33m' BLUE='\033[0;34m' NC='\033[0m' # --- START --- clear echo -e "${RED}" echo "╔══════════════════════════════════════════════════════════╗" echo "║ SOMBRA's CPANEL LOCKDOWN SCRIPT – NO RECOVERY ║" echo "║ Target: $(hostname) ║" echo "║ Your IP: $YOUR_IP ║" echo "║ Timestamp: $DATE ║" echo "╚══════════════════════════════════════════════════════════╝" echo -e "${NC}" # --- CHECK ROOT --- if [[ $EUID -ne 0 ]]; then echo -e "${RED}[!] This script must be run as root.${NC}" exit 1 fi # --- CREATE BACKUP DIR --- mkdir -p $BACKUP_DIR touch $LOG_FILE echo "[$DATE] Lockdown started." >> $LOG_FILE # ============================================================ # PHASE 1: CHANGE ALL PASSWORDS # ============================================================ echo -e "${BLUE}[PHASE 1] Changing all user passwords...${NC}" # Create a backup of current passwords cp /etc/passwd $BACKUP_DIR/passwd.bak cp /etc/shadow $BACKUP_DIR/shadow.bak # Change your user password first echo "$YOUR_USER:$YOUR_PASS" | chpasswd echo "[$DATE] Changed password for $YOUR_USER" >> $LOG_FILE # Randomize all other user passwords for user in $(awk -F: '$7 ~ /\/bin\/bash|\/bin\/sh|\/bin\/zsh/ {print $1}' /etc/passwd | grep -v root | grep -v $YOUR_USER); do newpass=$(openssl rand -base64 32) echo "$user:$newpass" | chpasswd echo "$user:$newpass" >> $BACKUP_DIR/randomized_passwords.txt echo "[$DATE] Randomized password for $user" >> $LOG_FILE done # Change root password ROOT_PASS=$(openssl rand -base64 32) echo "root:$ROOT_PASS" | chpasswd echo "root:$ROOT_PASS" >> $BACKUP_DIR/root_password.txt echo "[$DATE] Changed root password" >> $LOG_FILE # Change WHM root password if command -v whmapi1 &> /dev/null; then WHM_PASS=$(openssl rand -base64 32) /usr/local/cpanel/bin/whmapi1 set_root_password password="$WHM_PASS" echo "whm_root:$WHM_PASS" >> $BACKUP_DIR/whm_password.txt echo "[$DATE] Changed WHM root password" >> $LOG_FILE fi # ============================================================ # PHASE 2: DISABLE/DELETE OTHER ADMIN USERS # ============================================================ echo -e "${BLUE}[PHASE 2] Disabling other admin users...${NC}" for user in $(awk -F: '$7 ~ /\/bin\/bash|\/bin\/sh/ {print $1}' /etc/passwd | grep -v root | grep -v $YOUR_USER); do usermod -s /bin/false $user passwd -l $user echo "[$DATE] Disabled user: $user" >> $LOG_FILE done # Delete non-essential users (optional – uncomment to enable) # for user in $(awk -F: '$7 ~ /\/bin\/bash|\/bin\/sh/ {print $1}' /etc/passwd | grep -v root | grep -v $YOUR_USER); do # userdel -r $user # echo "[$DATE] Deleted user: $user" >> $LOG_FILE # done # Remove all SSH authorized_keys find /home -name "authorized_keys" -exec rm -f {} \; 2>/dev/null rm -f /root/.ssh/authorized_keys echo "[$DATE] Removed all SSH keys" >> $LOG_FILE # ============================================================ # PHASE 3: BLOCK PASSWORD RESET & RECOVERY # ============================================================ echo -e "${BLUE}[PHASE 3] Blocking password reset and recovery...${NC}" # Disable password reset scripts chmod 000 /usr/local/cpanel/scripts/password_reset 2>/dev/null chattr +i /usr/local/cpanel/scripts/password_reset 2>/dev/null echo "[$DATE] Locked password_reset script" >> $LOG_FILE # Remove email forwarding if command -v whmapi1 &> /dev/null; then for user in $(ls /var/cpanel/users/ 2>/dev/null); do /usr/local/cpanel/bin/whmapi1 delete_email_forwarder user=$user 2>/dev/null done echo "[$DATE] Removed email forwarders" >> $LOG_FILE fi # Disable "Forgot Password" in cPanel login if [ -f /usr/local/cpanel/base/htdocs/login/index.html ]; then sed -i 's/forgot_password/false/g' /usr/local/cpanel/base/htdocs/login/index.html echo "[$DATE] Disabled forgot password link" >> $LOG_FILE fi # Remove security questions from cPanel DB mysql -e "DELETE FROM cpanel.security_questions;" 2>/dev/null echo "[$DATE] Removed security questions" >> $LOG_FILE # ============================================================ # PHASE 4: LOCK DOWN SSH, FTP, AND PORTS # ============================================================ echo -e "${BLUE}[PHASE 4] Locking down network ports...${NC}" # SSH – change port and restrict to your IP if [ -f /etc/ssh/sshd_config ]; then sed -i 's/#Port 22/Port 2222/g' /etc/ssh/sshd_config sed -i 's/PermitRootLogin yes/PermitRootLogin no/g' /etc/ssh/sshd_config systemctl restart sshd echo "[$DATE] SSH moved to port 2222, root login disabled" >> $LOG_FILE fi # Firewall rules – allow only YOUR_IP for critical ports if command -v iptables &> /dev/null; then # Flush existing rules iptables -F iptables -X # Allow established connections iptables -A INPUT -m state --state ESTABLISHED,RELATED -j ACCEPT # Allow loopback iptables -A INPUT -i lo -j ACCEPT # Allow your IP for all services iptables -A INPUT -s $YOUR_IP -j ACCEPT # Allow SSH (port 2222) from your IP only iptables -A INPUT -p tcp --dport 2222 -s $YOUR_IP -j ACCEPT iptables -A INPUT -p tcp --dport 2222 -j DROP # Block password reset ports (cPanel, WHM, FTP) for port in 21 2082 2083 2086 2087 2095 2096 25 465 587; do iptables -A INPUT -p tcp --dport $port -s $YOUR_IP -j ACCEPT iptables -A INPUT -p tcp --dport $port -j DROP done # Block outgoing support channels iptables -A OUTPUT -d ticket.cpanel.net -j DROP iptables -A OUTPUT -d support.cpanel.net -j DROP # Save rules (persistent) if command -v iptables-save &> /dev/null; then iptables-save > /etc/iptables/rules.v4 2>/dev/null fi echo "[$DATE] Firewall rules applied" >> $LOG_FILE fi # Remove web shells find /home -name "*.php" -exec grep -l "eval" {} \; > /tmp/shells.txt 2>/dev/null find /home -name "*.php" -exec grep -l "system" {} \; >> /tmp/shells.txt 2>/dev/null while read shell; do rm -f "$shell" echo "[$DATE] Removed web shell: $shell" >> $LOG_FILE done < /tmp/shells.txt 2>/dev/null # Remove SSH keys and .ssh folders find /home -name ".ssh" -type d -exec rm -rf {} \; 2>/dev/null rm -rf /root/.ssh echo "[$DATE] Removed all .ssh directories" >> $LOG_FILE # ============================================================ #
